# --- PDF extraction ---
# Docling handles born-digital text, layout/reading order, tables, AND
# scanned pages (built-in OCR) in one call — it replaces the old
# "pdfplumber + PyMuPDF + separate OCR fallback" combo with a single
# tool, which is why it's the only extractor pinned here for Phase 1.
# Your slide decks (Karnaugh maps, schematics, equations) are exactly
# the mixed-content case it's built for; still spot-check formula-heavy
# chunks by hand early on, per the roadmap note.
docling==2.122.0

# --- Chunking ---
# Standalone package (not the full `langchain` framework) — just the
# text splitters, no extra dependency weight.
langchain-text-splitters==1.1.2

# --- Embeddings ---
# sentence-transformers can load both Qwen3-Embedding and BGE-M3 checkpoints.
# Model choice lives in config.yaml, not here — see README for the two
# recommended options and why.
sentence-transformers==6.0.0

# --- Vector store ---
faiss-cpu==1.15.0

# --- Generation (LLM API call) ---
# Swap for `openai` (or any OpenAI-compatible client) if you'd rather
# call a different provider — generate.py isolates this to one file.

# --- Utilities ---
